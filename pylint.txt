def carry_forward(rows: list[dict]) -> list[dict]:
    """Return rows with missing numeric values filled from previous row."""
    carried: list[dict] = []
    state: dict = {}
    for row in rows:
        state = {**state, **row}
        carried.append(dict(state))
    return carried
